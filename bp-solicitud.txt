<?php
// Recibe el formulario de la portada de bieripools.com y lo manda por correo.
header('Content-Type: application/json; charset=UTF-8');
if ($_SERVER['REQUEST_METHOD'] !== 'POST') { http_response_code(405); echo json_encode(['ok'=>false]); exit; }
if (!empty($_POST['web'])) { echo json_encode(['ok'=>true]); exit; } // trampa anti-spam
$c = function($k,$max=500){ return htmlspecialchars(mb_substr(trim($_POST[$k] ?? ''),0,$max), ENT_QUOTES, 'UTF-8'); };
$nombre=$c('n',120); $tel=$c('t',40); $ciudad=$c('c',120); $tipo=$c('s',80); $msg=$c('m',2000); $pref=$c('p',20);
if ($nombre==='' || $tel==='') { http_response_code(400); echo json_encode(['ok'=>false]); exit; }
$para = 'info@bieripools.com';
$asunto = '=?UTF-8?B?'.base64_encode(($pref==='Llamada' ? 'LLAMAR: ' : 'WhatsApp: ')."Solicitud de cotización de $nombre").'?=';
$cuerpo = "<h3 style='color:#0E2A47'>Nueva solicitud desde la portada de bieripools.com</h3>"
 ."<p><b>Contactar por:</b> ".($pref==='Llamada' ? 'Llamada telefónica (el cliente pide que le llamemos)' : 'WhatsApp')."</p>"
 ."<p><b>Nombre:</b> $nombre</p><p><b>Teléfono / WhatsApp:</b> <a href='tel:$tel'>$tel</a></p>"
 ."<p><b>Ciudad:</b> $ciudad</p><p><b>Tipo de proyecto:</b> $tipo</p><p><b>Su alberca:</b><br>".nl2br($msg)."</p>"
 ."<hr><small>Enviado el ".date('d/m/Y H:i')." desde el formulario de la portada.</small>";
$cab  = "MIME-Version: 1.0\r\nContent-type: text/html; charset=UTF-8\r\n";
$cab .= "From: Bieri Pools Web <info@bieripools.com>\r\n";
$cab .= "Cc: direccion@bieripools.com\r\n";
$ok = mail($para, $asunto, $cuerpo, $cab);
echo json_encode(['ok'=>(bool)$ok]);
